# %% CELL 2: Import libraries
import os
import tarfile
import pickle
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

from pathlib import Path
from sklearn.preprocessing import LabelBinarizer
from sklearn.metrics import classification_report

# %% CELL 3: Extract uploaded CIFAR-10 dataset
archive = Path.cwd() / "cifar-10-python.tar.gz"
extract_dir = Path.cwd() / "cifar_extracted"

if not archive.exists():
    raise FileNotFoundError("Upload cifar-10-python.tar.gz using Jupyter Upload first.")

extract_dir.mkdir(exist_ok=True)

batch_dir = extract_dir / "cifar-10-batches-py"

if not batch_dir.exists():
    with tarfile.open(archive, "r:gz") as tar:
        tar.extractall(extract_dir)

print("Dataset extracted:", batch_dir.exists())

# %% CELL 4: Load CIFAR-10 batches
def load_batch(file_path):
    with open(file_path, "rb") as f:
        data = pickle.load(f, encoding="bytes")
    images = data[b"data"]
    labels = data[b"labels"]
    images = images.reshape(-1, 3, 32, 32).transpose(0, 2, 3, 1)
    return images, labels

x_train_list = []
y_train_list = []

for i in range(1, 6):
    x, y = load_batch(batch_dir / f"data_batch_{i}")
    x_train_list.append(x)
    y_train_list.extend(y)

x_train = np.concatenate(x_train_list)
y_train = np.array(y_train_list)

x_test, y_test = load_batch(batch_dir / "test_batch")
y_test = np.array(y_test)

print("Training data:", x_train.shape)
print("Testing data:", x_test.shape)

# %% CELL 5: Preprocess data
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

x_train = x_train.reshape(len(x_train), -1)
x_test = x_test.reshape(len(x_test), -1)

encoder = LabelBinarizer()
y_train_onehot = encoder.fit_transform(y_train)
y_test_onehot = encoder.transform(y_test)

print("Flattened training data:", x_train.shape)

# %% CELL 6: Define Feedforward Neural Network
model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(3072,)),
    tf.keras.layers.Dense(256, activation="sigmoid"),
    tf.keras.layers.Dense(128, activation="sigmoid"),
    tf.keras.layers.Dense(10, activation="softmax")
])

model.summary()

# %% CELL 7: Compile model using SGD
model.compile(
    optimizer=tf.keras.optimizers.SGD(
        learning_rate=0.01,
        momentum=0.9
    ),
    loss="categorical_crossentropy",
    metrics=["accuracy"]
)

# %% CELL 8: Train model
history = model.fit(
    x_train,
    y_train_onehot,
    epochs=3,
    batch_size=128,
    validation_split=0.1
)

# %% CELL 9: Evaluate model
test_loss, test_accuracy = model.evaluate(
    x_test, y_test_onehot, verbose=0
)

print("Test Loss:", test_loss)
print("Test Accuracy:", test_accuracy)

# %% CELL 10: Classification report
predictions = np.argmax(model.predict(x_test, verbose=0), axis=1)

print(classification_report(
    y_test,
    predictions,
    target_names=[
        "airplane", "automobile", "bird", "cat", "deer",
        "dog", "frog", "horse", "ship", "truck"
    ]
))

# %% CELL 11: Plot accuracy and loss
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(history.history["accuracy"], label="Training")
plt.plot(history.history["val_accuracy"], label="Validation")
plt.title("Model Accuracy")
plt.xlabel("Epoch")
plt.legend()

plt.subplot(1, 2, 2)
plt.plot(history.history["loss"], label="Training")
plt.plot(history.history["val_loss"], label="Validation")
plt.title("Model Loss")
plt.xlabel("Epoch")
plt.legend()

plt.tight_layout()
plt.show()
