# %% CELL 2: Import libraries
import tarfile
import pickle
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

from pathlib import Path
from sklearn.metrics import classification_report

# %% CELL 3: Extract uploaded dataset if necessary
archive = Path.cwd() / "cifar-10-python.tar.gz"
extract_dir = Path.cwd() / "cifar_extracted"
batch_dir = extract_dir / "cifar-10-batches-py"

if not archive.exists():
    raise FileNotFoundError("Upload cifar-10-python.tar.gz first.")

if not batch_dir.exists():
    extract_dir.mkdir(exist_ok=True)
    with tarfile.open(archive, "r:gz") as tar:
        tar.extractall(extract_dir)

# %% CELL 4: Load CIFAR-10
def load_batch(file_path):
    with open(file_path, "rb") as f:
        data = pickle.load(f, encoding="bytes")

    images = data[b"data"]
    labels = data[b"labels"]

    images = images.reshape(-1, 3, 32, 32)
    images = images.transpose(0, 2, 3, 1)

    return images, labels

x_train_list = []
y_train_list = []

for i in range(1, 6):
    x, y = load_batch(batch_dir / f"data_batch_{i}")
    x_train_list.append(x)
    y_train_list.extend(y)

x_train = np.concatenate(x_train_list)
y_train = np.array(y_train_list)

x_test, y_test = load_batch(batch_dir / "test_batch")
y_test = np.array(y_test)

print("Training images:", x_train.shape)
print("Testing images:", x_test.shape)

# %% CELL 5: Normalize images
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

class_names = [
    "airplane", "automobile", "bird", "cat", "deer",
    "dog", "frog", "horse", "ship", "truck"
]

# %% CELL 6: Display sample image
plt.imshow(x_train[0])
plt.title(class_names[int(y_train[0])])
plt.axis("off")
plt.show()

# %% CELL 7: Define CNN architecture
model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(32, 32, 3)),

    tf.keras.layers.Conv2D(32, (3, 3), activation="relu"),
    tf.keras.layers.MaxPooling2D((2, 2)),

    tf.keras.layers.Conv2D(64, (3, 3), activation="relu"),
    tf.keras.layers.MaxPooling2D((2, 2)),

    tf.keras.layers.Flatten(),
    tf.keras.layers.Dense(64, activation="relu"),
    tf.keras.layers.Dense(10, activation="softmax")
])

model.summary()

# %% CELL 8: Compile model
model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

# %% CELL 9: Train CNN
history = model.fit(
    x_train,
    y_train,
    epochs=3,
    batch_size=128,
    validation_split=0.1
)

# %% CELL 10: Evaluate model
test_loss, test_accuracy = model.evaluate(
    x_test, y_test, verbose=0
)

print("Test Loss:", test_loss)
print("Test Accuracy:", test_accuracy)

# %% CELL 11: Classification report
predictions = np.argmax(model.predict(x_test, verbose=0), axis=1)

print(classification_report(
    y_test,
    predictions,
    target_names=class_names
))

# %% CELL 12: Plot training performance
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(history.history["accuracy"], label="Training")
plt.plot(history.history["val_accuracy"], label="Validation")
plt.title("CNN Accuracy")
plt.xlabel("Epoch")
plt.legend()

plt.subplot(1, 2, 2)
plt.plot(history.history["loss"], label="Training")
plt.plot(history.history["val_loss"], label="Validation")
plt.title("CNN Loss")
plt.xlabel("Epoch")
plt.legend()

plt.tight_layout()
plt.show()

# %% CELL 13: Predict one image
index = 10

prediction = model.predict(
    x_test[index:index+1], verbose=0
)

predicted_class = np.argmax(prediction)

plt.imshow(x_test[index])
plt.title("Predicted: " + class_names[predicted_class])
plt.axis("off")
plt.show()

print("Actual:", class_names[int(y_test[index])])
print("Predicted:", class_names[predicted_class])
